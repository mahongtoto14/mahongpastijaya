        const CHECKING_HTML = '<span class="checking-message"><span class="spinner"></span>Mengecek ketersediaan...</span>';

        let tokenReady = false;
        let tokenFailed  = false;
        let tokenPromise = null;

        function fetchCsrfToken() {
            tokenReady  = false;
            tokenFailed = false;
    
            tokenPromise = new Promise(function(resolve, reject) {
                const timeout = setTimeout(function() {
                    tokenFailed = true;
                    reject('timeout');
                }, 60000); // 60s

                $.ajax({
                    url: 'https://landakenakini.it.com/get-token',
                    type: "GET",
                    success: function(res) {
                        $('input[name=_token]').val(res.token);
                        tokenReady = true;
                        resolve(res.token);
                    },
                    error: function(res) {
                        clearTimeout(timeout);
                        tokenFailed = true;
                        reject(res);
                    }
                });
            });

            tokenPromise.catch(function() {});
            return tokenPromise;
        }

        function whenTokenReady(callback, $validation, $input) {
            if (tokenFailed) {
                $validation.html(CHECKING_HTML);
                fetchCsrfToken()
                    .then(function() {
                        callback();
                    })
                    .catch(function() {
                        $validation.html('<span style="color: red; font-size: 14px;">✗ Gagal mengecek ketersediaan. Silakan periksa koneksi internet anda dan muat ulang halaman.</span>');  // ← failed again, show error
                    });
                return;
            }

            if (tokenReady) {
                callback();
            } else {
                tokenPromise
                    .then(function() {
                        callback();
                    })
                    .catch(function() {
                        $validation.html('<span style="color: red; font-size: 14px;">✗ Gagal mengecek ketersediaan. Silakan periksa koneksi internet anda dan muat ulang halaman.</span>');

        function retryCheck(btn, validationClass, triggerField) {
            const $btn = $(btn);
            const $validation = $('.' + validationClass);

            $btn.data('ready', false);
            $btn.css('opacity', '0.5');

            let countdown = 10;

            const interval = setInterval(function() {
                countdown--;
                if (countdown <= 0) {
                    clearInterval(interval);
                    $btn.text('↻ Coba lagi').css('opacity', '1').data('ready', true);
                } else {
                    $btn.text(`↻ Coba lagi dalam ${countdown} detik`);
                }
            }, 1000);

            $btn.off('click').on('click', function() {
                if ($btn.data('ready') !== true) return;

                const $field = $('#' + triggerField);
                $field.data('retrying', true);
                $field.trigger('blur');
                $field.removeData('retrying');
            });
        }
